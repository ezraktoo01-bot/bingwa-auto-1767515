// API - SMS kutoka SMS Forwarder
app.post('/api/mpesa-sms', async (req, res) => {
  try {
    const rawBody = JSON.stringify(req.body);
    const bodyLower = rawBody.toLowerCase();
    console.log("📩 SMS IN:", bodyLower.substring(0,300));

    // FILTER - TILL PEKEE - Personal 0725 ignore
    if(!bodyLower.includes("1767515") && !bodyLower.includes("till number") && !bodyLower.includes("till")){
      console.log("❌ IGNORED - ni ya personal "+PERSONAL_LINE+" sio Till");
      return res.json({ status: "ignored", reason: "personal line - not Till 1767515", till: TILL });
    }

    const smsText = req.body.message || req.body.text || req.body.body || req.body.sms || rawBody;
    
    // Extract Amount
    let amountMatch = smsText.match(/Ksh\s*(\d+)[.,]?(\d+)?/i) || smsText.match(/\b(19|49|99|199)\b/);
    let amount = amountMatch ? parseInt(amountMatch[1]) : null;

    // Extract Phone - 07xx au 2547xx
    let phoneMatch = smsText.match(/(?:from|07)\s*(0?7\d{8})/i) || smsText.match(/\b(07\d{8})\b/) || smsText.match(/\b(2547\d{8})\b/);
    let phone = phoneMatch ? phoneMatch[1] : null;
    if(phone){
      if(phone.startsWith("0")) phone = "254" + phone.substring(1);
      if(phone.startsWith("7")) phone = "254" + phone;
      if(!phone.startsWith("254")) phone = "254" + phone;
    }

    // Fallback - tafuta namba yoyote ya 07
    if(!phone){
      const fallback = smsText.match(/07\d{8}/);
      if(fallback){
        phone = fallback[0];
        if(phone.startsWith("0")) phone = "254" + phone.substring(1);
      }
    }

    console.log(`💰 PARSED: ${amount} KES from ${phone}`);

    if(!amount || !phone){
      TRANSACTIONS.unshift({ time: new Date().toISOString(), raw: smsText.substring(0,200), amount, phone, status: "parse_failed" });
      return res.json({ status: "parse_failed", raw: smsText, hint: "hakuna amount au phone" });
    }

    // Map package
    let pkg = PACKAGES[amount.toString()];
    if(!pkg){
      if(amount >= 15 && amount <= 25) pkg = PACKAGES["19"];
      else if(amount >= 40 && amount <= 60) pkg = PACKAGES["49"];
      else if(amount >= 85 && amount <= 115) pkg = PACKAGES["99"];
      else if(amount >= 180 && amount <= 220) pkg = PACKAGES["199"];
    }

    if(!pkg){
      TRANSACTIONS.unshift({ time: new Date().toISOString(), amount, phone, status: "unsupported_amount" });
      return res.json({ status: "unsupported", amount, supported: Object.keys(PACKAGES) });
    }

    // SEND!
    const sendResult = await sendBundle(phone, pkg);
    
    const tx = {
      id: Date.now(),
      time: new Date().toISOString(),
      till: TILL,
      amount,
      phone,
      package: pkg.name,
      data: pkg.data,
      duration: pkg.duration,
      result: sendResult,
      raw: smsText.substring(0,200)
    };
    TRANSACTIONS.unshift(tx);
    if(TRANSACTIONS.length > 100) TRANSACTIONS.pop();

    console.log(`✅ DONE: ${pkg.name} to ${phone} - ${sendResult.time}`);

    res.json({
      status: "success",
      till: TILL,
      personal_filtered: PERSONAL_LINE,
      amount,
      phone,
      package: pkg.name,
      delivered_in: sendResult.time,
      mode: sendResult.provider,
      at_configured: !!process.env.AT_API_KEY
    });

  } catch(e){
    console.error("ERROR:", e);
    res.status(500).json({ error: e.message });
  }
});
